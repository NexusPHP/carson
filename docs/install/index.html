Workflow file

      
Save as .github/workflows/carson.yml in each repository where Carson should run. Trim triggers you do not need.

      
name: Carson

on:
  pull_request_target:
    types: [opened, synchronize, reopened, edited]
  pull_request_review:
    types: [submitted]
  push:
  issues:
    types: [opened, edited]
  issue_comment:
    types: [created]
  schedule:
    - cron: '0 0 * * *'

jobs:
  carson:
    runs-on: ubuntu-latest
    steps:
      - uses: NexusPHP/carson@<commit-sha> # v1.0.0
        with:
          app_id: ${{ secrets.CARSON_APP_ID }}
          private_key: ${{ secrets.CARSON_PRIVATE_KEY }}
